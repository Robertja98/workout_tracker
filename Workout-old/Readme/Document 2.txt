## **Starter Code: app.py**

```python
from flask import Flask, request, jsonify, render_template
import csv
import os
from utils import add_workout, read_workouts

app = Flask(__name__)
CSV_FILE = 'data/workouts.csv'

# Ensure CSV file exists
if not os.path.exists(CSV_FILE):
    with open(CSV_FILE, 'w', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(['date', 'exercise', 'reps', 'weight'])

@app.route('/')
def home():
    return render_template('dashboard.html', workouts=read_workouts(CSV_FILE))

@app.route('/add', methods=['POST'])
def add():
    data = request.get_json()
    if not data:
        return jsonify({'error': 'Invalid JSON'}), 400
    add_workout(CSV_FILE, [data['date'], data['exercise'], data['reps'], data['weight']])
    return jsonify({'status': 'success', 'message': 'Workout added'})

@app.route('/dashboard')
def dashboard():
    workouts = read_workouts(CSV_FILE)
    return render_template('dashboard.html', workouts=workouts)

if __name__ == '__main__':
    app.run(debug=True)
```

## **Helper Functions: utils.py**

```python
import csv

def add_workout(csv_file, data):
    with open(csv_file, 'a', newline='') as f:
        writer = csv.writer(f)
        writer.writerow(data)

def read_workouts(csv_file):
    with open(csv_file, 'r') as f:
        reader = csv.reader(f)
        return list(reader)[1:]  # Skip header
```
